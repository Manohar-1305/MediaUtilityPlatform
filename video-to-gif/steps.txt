# Build & Run the container
docker build -t video-to-gif:latest . && docker rm -f video-to-gif 2>/dev/null; docker run -d --name video-to-gif --restart unless-stopped --env-file .env --network media-tools-network -p 5014:5014 video-to-gif:latest

# Verify Redis 
docker exec video-to-gif python -c 'from redis_cache import redis_client; print("Redis:",redis_client.ping()); keys=redis_client.keys("media-tools:video-to-gif:conversion:*"); print("video-to-gif cache entries:",len(keys)); [print(k.decode(),redis_client.strlen(k),"bytes") for k in keys]'
