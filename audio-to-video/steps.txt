# Build and recreate the audio-to-video container using .env
docker build -t audio-to-video . && (docker rm -f audio-to-video 2>/dev/null || true) && docker run -d --name audio-to-video --network media-tools-network --restart unless-stopped --env-file ~/MediaUtilityPlatform/audio-cutter/.env -e PORT=5007 -p 5007:5007 audio-to-video

# Verify Redis connectivity and import Redis cache hooks
docker exec audio-to-video python -c 'import redis_cache_hooks,redis,os; r=redis.Redis(host=os.getenv("REDIS_HOST"),port=int(os.getenv("REDIS_PORT")),db=int(os.getenv("REDIS_DB")),password=os.getenv("REDIS_PASSWORD")); print("Redis cache hooks imported successfully"); print("Redis:",r.ping()); print("Keys:",r.keys("*"))'

# Check the cache in redis
docker exec redis redis-cli -a 'Redis123!' --no-auth-warning EVAL "local r={}; for _,k in ipairs(redis.call('keys','media-tools:conversion:*')) do local v=redis.call('get',k); if v then table.insert(r,{k,string.sub(v,1,100)}) end end; return r" 0
