
# Build Docker
docker build -t image-to-pdf . && (docker rm -f image-to-pdf 2>/dev/null || true) && docker run -d --name image-to-pdf --network media-tools-network --restart unless-stopped --env-file ~/MediaUtilityPlatform/audio-cutter/.env -e PORT=5008 -p 5008:5008 image-to-pdf

# Import Redis
docker exec image-to-pdf sh -c 'python -c "import redis; r=redis.Redis(host=\"redis\",port=6379,password=\"Redis123!\",decode_responses=True); print(\"Redis connected:\",r.ping())"'

# Import Redis caching
docker exec image-to-pdf python -c 'import redis_cache_hooks,redis,os; r=redis.Redis(host=os.getenv("REDIS_HOST"),port=int(os.getenv("REDIS_PORT")),db=int(os.getenv("REDIS_DB")),password=os.getenv("REDIS_PASSWORD")); print("Redis cache hooks imported successfully"); print("Redis:",r.ping()); print("Keys:",r.keys("*"))'

# Check if it is caching
docker exec image-to-pdf python -c 'import redis_cache_hooks; print("Redis cache hooks imported successfully")'

docker logs -f image-to-pdf

docker ps --filter name=image-to-pdf
