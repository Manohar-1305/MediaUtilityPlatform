
# Build & run Container
docker build --no-cache -t video-cutter:latest . && docker run -d --name video-cutter --network media-tools-network --restart unless-stopped --env-file .env -e PORT=5003 -p 5003:5003 video-cutter:latest


# Enable redis cache and verify
docker exec video-cutter python -c 'import redis_cache,redis_cache_hooks,redis,os; r=redis.Redis(host=os.getenv("REDIS_HOST","redis"),port=int(os.getenv("REDIS_PORT","6379")),db=int(os.getenv("REDIS_DB","0")),password=os.getenv("REDIS_PASSWORD") or None); print("Redis cache imports successful"); print("Redis connected:",r.ping()); print("Keys:",r.keys("*"))'
